    <script class="">
      // Simple star rating interaction
        const starBtns = document.querySelectorAll('.star-btn');
        const starIcons = document.querySelectorAll('.star-icon');
        let currentRating = 0;

        starBtns.forEach(btn => {
            btn.addEventListener('click', (e) => {
                const value = parseInt(btn.dataset.value);
                currentRating = value;
                
                starIcons.forEach((icon, index) => {
                    if (index < value) {
                        icon.classList.remove('text-outline-variant');
                        icon.classList.add('text-[#eab308]'); // Yellow warning/star color
                    } else {
                        icon.classList.remove('text-[#eab308]');
                        icon.classList.add('text-outline-variant');
                    }
                });
            });
        });

        // Set default rating to 5 as requested in prompt "Rating prompt: 5 stars"
        document.querySelector('.star-btn[data-value="5"]').click();
